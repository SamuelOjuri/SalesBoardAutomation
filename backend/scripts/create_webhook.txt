$secureToken = Read-Host "Paste the app OAuth access token" -AsSecureString
$credential = New-Object System.Management.Automation.PSCredential (
    "unused",
    $secureToken
)
$appAccessToken = $credential.GetNetworkCredential().Password

$mutation = @'
mutation CreateEmailFileWebhook(
  $boardId: ID!
  $url: String!
  $config: JSON!
) {
  create_webhook(
    board_id: $boardId
    url: $url
    event: change_specific_column_value
    config: $config
  ) {
    id
    board_id
    event
    config
  }
}
'@

$requestBody = @{
    query = $mutation
    variables = @{
        boardId = "5100711564"
        url = "https://salesboardautomation.onrender.com/api/monday/webhooks"
        config = '{"columnId":"file_mm5erpbb"}'
    }
} | ConvertTo-Json -Depth 10

$response = Invoke-RestMethod `
    -Method Post `
    -Uri "https://api.monday.com/v2" `
    -Headers @{
        Authorization = $appAccessToken
        "API-Version" = "2026-07"
    } `
    -ContentType "application/json" `
    -Body $requestBody

if ($response.errors) {
    $response.errors | ConvertTo-Json -Depth 10
    throw "monday returned GraphQL errors"
}

$response.data.create_webhook | Format-List

$appAccessToken = $null
$credential = $null
$secureToken.Dispose()
